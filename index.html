End-to-End Wear Scar Detection & Training Guide

Install dependencies, annotate wear scar images with LabelMe, convert JSON annotations to YOLO format, configure your dataset and train a large-scale YOLO26 model (yolo26l.pt).

YOLO26 LabelMe class: scar 80 / 20 split
Install
Open LabelMe
Draw ROI
Save JSON
Convert → YOLO
1

Install Dependencies

Open your terminal or command prompt and install all required Python libraries:

pip install ultralytics labelme scikit-learn opencv-python
2

Open LabelMe

Launch the graphical annotation interface from your terminal:

labelme

Click "Open Dir" and select the folder containing your raw microscope wear-scar images (.jpg / .png).

3

Draw ROI (Region of Interest)

  • Click "Create Polygon" in the sidebar (or press Ctrl+N).
  • Click points to draw a closed polygon tracing the exact visible boundary of the wear scar.
  • When prompted for a label, type scar and click OK.
4

Save in Proper Directory Format

Click Save (or Ctrl+S). LabelMe writes a .json with the same name as the image.

Crucial rule: keep each image and its JSON together, e.g. specimen_01.jpg + specimen_01.json.
raw_labelme_data/
specimen_01.jpg
specimen_01.json
…
5

Convert LabelMe JSON → YOLO Format

Save as convert.py in your project folder, then run python convert.py

Input./raw_labelme_data
Output./dataset/images|labels/{train,val}
Label formatclass x_c y_c w h (normalised)
import os
import json
import shutil
from sklearn.model_selection import train_test_split

def convert_labelme_to_yolo(data_dir, output_dir, test_size=0.2):
    images_dir = os.path.join(output_dir, "images")
    labels_dir = os.path.join(output_dir, "labels")

    for split in ["train", "val"]:
        os.makedirs(os.path.join(images_dir, split), exist_ok=True)
        os.makedirs(os.path.join(labels_dir, split), exist_ok=True)

    json_files = [f for f in os.listdir(data_dir) if f.endswith('.json')]
    
    if not json_files:
        print(f"No JSON files found in {data_dir}!")
        return

    train_files, val_files = train_test_split(json_files, test_size=test_size, random_state=42)
    dataset_splits = {"train": train_files, "val": val_files}

    class_mapping = {"scar": 0}

    for split, files in dataset_splits.items():
        print(f"Processing {split} split ({len(files)} files)...")
        for json_file in files:
            json_path = os.path.join(data_dir, json_file)
            with open(json_path, 'r', encoding='utf-8') as f:
                data = json.load(f)

            img_width = data.get("imageWidth")
            img_height = data.get("imageHeight")
            image_path_field = data.get("imagePath")

            base_name = os.path.splitext(json_file)[0]
            img_ext = os.path.splitext(image_path_field)[1] if image_path_field else ".jpg"
            src_img_path = os.path.join(data_dir, base_name + img_ext)

            if not os.path.exists(src_img_path):
                for ext in ['.jpg', '.JPG', '.png', '.PNG', '.jpeg']:
                    alt_path = os.path.join(data_dir, base_name + ext)
                    if os.path.exists(alt_path):
                        src_img_path = alt_path
                        break

            if not os.path.exists(src_img_path):
                print(f"Warning: Image for {json_file} not found. Skipping.")
                continue

            dst_img_path = os.path.join(images_dir, split, base_name + os.path.splitext(src_img_path)[1])
            shutil.copy(src_img_path, dst_img_path)

            txt_path = os.path.join(labels_dir, split, base_name + ".txt")
            with open(txt_path, 'w', encoding='utf-8') as txt_file:
                for shape in data.get("shapes", []):
                    label = shape.get("label")
                    if label not in class_mapping:
                        continue
                    class_id = class_mapping[label]
                    points = shape.get("points")

                    if shape.get("shape_type") == "polygon":
                        x_coords = [p[0] for p in points]
                        y_coords = [p[1] for p in points]
                        xmin, xmax = min(x_coords), max(x_coords)
                        ymin, ymax = min(y_coords), max(y_coords)

                        xmin = max(0.0, min(xmin, img_width))
                        xmax = max(0.0, min(xmax, img_width))
                        ymin = max(0.0, min(ymin, img_height))
                        ymax = max(0.0, min(ymax, img_height))

                        b_width = xmax - xmin
                        b_height = ymax - ymin
                        b_x_center = xmin + (b_width / 2.0)
                        b_y_center = ymin + (b_height / 2.0)

                        x_norm = b_x_center / img_width
                        y_norm = b_y_center / img_height
                        w_norm = b_width / img_width
                        h_norm = b_height / img_height

                        txt_file.write(f"{class_id} {x_norm:.6f} {y_norm:.6f} {w_norm:.6f} {h_norm:.6f}\n")

    print(f"Dataset conversion complete! Output saved to: {output_dir}")

if __name__ == "__main__":
    RAW_ANNOTATIONS_DIR = "./raw_labelme_data"
    PROCESSED_DATA_DIR = "./dataset"
    convert_labelme_to_yolo(RAW_ANNOTATIONS_DIR, PROCESSED_DATA_DIR, test_size=0.2)
